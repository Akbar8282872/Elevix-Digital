        class TopWavesPlugin {
            constructor(canvasId) {
                this.canvas = document.getElementById(canvasId);
                if (!this.canvas) return;
                this.ctx = this.canvas.getContext('2d');
                this.time = 0;
                
                // Layers matching the screenshot: deep dark bodies with red glowing bottom edges

                    { speed: 0.0012, amplitude: 170, yOffset: 400, edgeColor: 'rgba(90, 5, 5, 0.3)' },
                    { speed: 0.0018, amplitude: 220, yOffset: 520, edgeColor: 'rgba(40, 0, 0, 0.15)' }     // Back-most
                ];
                
                window.addEventListener('resize', () => this.resize());
                this.resize();
                this.animate();
            }
            
            resize() {
                this.width = this.canvas.width = window.innerWidth;
                this.height = this.canvas.height = window.innerHeight;
            }
            
            animate() {
                this.ctx.clearRect(0, 0, this.width, this.height);
                
                // Draw layers from deepest (largest yOffset) to front-most (smallest yOffset)
                const sortedLayers = [...this.layers].sort((a, b) => b.yOffset - a.yOffset);

                        this.ctx.lineTo(x, y);

                    // Main slow waves
                    float elevation = snoise(vec3(pos.xy * 0.003, uTime * 0.12)) * 40.0;
                    // Secondary rapid ripples
                    elevation += snoise(vec3(pos.xy * 0.008 + vec2(10.0), uTime * 0.22)) * 12.0;
                    
                    // Mouse interactive ripple
                    // Map mouse position into plane coords
                    vec2 mousePos = uMouse * vec2(400.0, 300.0);
                    float dist = distance(pos.xy, mousePos);
                    if (dist < 400.0) {
                        float strength = (1.0 - dist / 400.0) * 35.0;
                        elevation += sin(uTime * 3.0 - dist * 0.03) * strength;
                    }
                    
                    pos.z += elevation;
                    vElevation = elevation;
                    
                    gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
                }
            `;

            // Custom GLSL Fragment Shader mapping height to glowing red/black gradient
            const fragmentShader = `
                varying vec2 vUv;
                varying float vElevation;
                uniform float uTime;

                void main() {
                    // Normalize elevation range [-40.0, 52.0] into [0, 1]
                    float mixStrength = (vElevation + 40.0) / 92.0;
                    mixStrength = clamp(mixStrength, 0.0, 1.0);

                    // Color palette
                    vec3 valleyColor = vec3(0.004, 0.0, 0.0);   // Dark black-red valley
                    vec3 midColor = vec3(0.45, 0.0, 0.01);      // Deep crimson (#8b0000 style)
                    vec3 peakColor = vec3(0.98, 0.02, 0.05);    // Glowing crimson red (#ff0000 style)

                    vec3 finalColor = vec3(0.0);
                    if (mixStrength < 0.35) {
                        finalColor = mix(valleyColor, midColor, mixStrength / 0.35);
                    } else {
                        finalColor = mix(midColor, peakColor, (mixStrength - 0.35) / 0.65);
                    }

                    // Height-dependent glow
                    float glow = pow(mixStrength, 3.2) * 0.35;
                    finalColor += vec3(glow, 0.0, 0.0);

                    // High-tech subtle digital grid overlay
                    float grid = abs(sin(vUv.x * 250.0) * sin(vUv.y * 250.0));
                    grid = step(0.988, grid);
                    finalColor = mix(finalColor, peakColor, grid * 0.12 * mixStrength);

                    gl_FragColor = vec4(finalColor, 1.0);
                }
            `;

            const geometry = new THREE.PlaneGeometry(1600, 1600, 160, 160);
            const material = new THREE.ShaderMaterial({
                vertexShader: vertexShader,
                fragmentShader: fragmentShader,
                uniforms: {
                    uTime: { value: 0 },
                    uMouse: { value: new THREE.Vector2(0, 0) }
                },
                wireframe: false,
                transparent: true
            });

            const mesh = new THREE.Mesh(geometry, material);
            // Tilt plane back so it slopes towards camera
            mesh.rotation.x = -Math.PI / 2.3;
            scene.add(mesh);

            // Time & Mouse variables
            let time = 0;
            const mouse = new THREE.Vector2(0, 0);
            const targetMouse = new THREE.Vector2(0, 0);

            document.addEventListener('mousemove', (event) => {
                // Normalize mouse position between -1.0 and 1.0
                targetMouse.x = (event.clientX / window.innerWidth) * 2 - 1;
                targetMouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
            });

            function resize() {
                const width = window.innerWidth;
                const height = window.innerHeight;
                
                renderer.setSize(width, height);
                camera.aspect = width / height;
                camera.updateProjectionMatrix();
            }
            window.addEventListener('resize', resize);

            function animate() {
                requestAnimationFrame(animate);

                time += 0.08;
                material.uniforms.uTime.value = time;

                // Lerp mouse coordinates for smooth animation
                mouse.x += (targetMouse.x - mouse.x) * 0.05;
                mouse.y += (targetMouse.y - mouse.y) * 0.05;
                material.uniforms.uMouse.value.copy(mouse);

                // camera tilt parallax based on mouse
                camera.position.x += (mouse.x * 40 - camera.position.x) * 0.03;
                camera.position.y += (-120 + mouse.y * 30 - camera.position.y) * 0.03;
                camera.lookAt(0, 40, 0);

                renderer.render(scene, camera);
            }

            animate();
        })();
